مثال 1: صنف Point (نقطة)

📋 تحليل المتطلبات

نريد تمثيل نقطة في المستوى ثنائي الأبعاد:

📊

الخصائص (Attributs)

x: الإحداثي الأفقي (double)

⚡

السلوكيات (Méthodes)

afficher(): عرض الإحداثيات

💻 الكود الكامل للصنف Point:

Point.java 
public class Point {
    // الخصائص (Attributs)
    private double x;
    private double y;
    
    // المنشئ الافتراضي (Constructeur par défaut)
    public Point() {
        this.x = 0;
        this.y = 0;
    }
    
    // المنشئ ذو المعاملات (Constructeur paramétré)
    public Point(double x, double y) {
        this.x = x;
        this.y = y;
    }
    
    // طرق الوصول (Accesseurs - Getters)
    public double getX() { return this.x; }
    public double getY() { return this.y; }
    
    // طرق التعديل (Mutateurs - Setters)
    public void setX(double x) { this.x = x; }
    public void setY(double y) { this.y = y; }
    
    // طريقة النقل
    public void deplacer(double a, double b) {
        this.x += a;
        this.y += b;
    }
    
    // طريقة العرض
    public void afficher() {
        System.out.println("(" + x + ", " + y + ")");
    }
}

مثال 2: صنف Cercle (دائرة)

Cercle.java 
public class Cercle {
    // خصائص الدائرة
    private Point centre;
    private double rayon;
    private String couleur;
    
    // المنشئ الأول
    public Cercle() {
        this.centre = new Point(0.0, 0.0);
        this.rayon = 2.0;
        this.couleur = "rouge";
    }
    
    // المنشئ الثاني (التزيين - Surcharge)
    public Cercle(double r, String c) {
        this.centre = new Point(1.0, 0.0);
        this.rayon = r;
        this.couleur = c;
    }
    
    // طريقة النقل
    public void deplacer(double a, double b) {
        this.centre.deplacer(a, b);
    }
    
    // حساب المساحة
    public double surface() {
        return Math.PI * rayon * rayon;
    }
}

التزيين (Overloading) يسمح بتعريف عدة طرق بنفس الاسم في نفس الصنف، بشرط اختلاف المعاملات (عددها أو نوعها).

  • يمكن تطبيق التزيين على المنشئات والطرق العادية
  • المترجم يحدد الطريقة المناسبة عند الاستدعاء

Point.java 
public class Point {
    // الخصائص
    private double x;
    private double y;

    // Getters
    public double getX() { return x; }
    public double getY() { return y; }

    // Setters
    public void setX(double x) { this.x = x; }
    public void setY(double y) { this.y = y; }

    // عرض الإحداثيات
    public void afficher() {
        System.out.println("(" + x + ", " + y + ")");
    }

    // حساب النقطة الوسطى
    public Point milieu(Point p) {
        Point m = new Point();
        m.setX((x + p.x) / 2);
        m.setY((y + p.y) / 2);
        return m;
    }

    // نظير النقطة بالنسبة للأصل
    public Point symetrie() {
        return new Point(-x, -y);
    }

    // اختبار التساوي
    public boolean egal(Point p) {
        return (x == p.x && y == p.y);
    }

    // المقارنة بالقرب من المركز
    public Point procheDuCentre(Point p) {
        double dist1 = Math.sqrt(x*x + y*y);
        double dist2 = Math.sqrt(p.x*p.x + p.y*p.y);
        if (dist1 <= dist2)
            return this;
        else
            return p;
    }

    // المنشئ الصريح (يُضاف لاحقاً)
    public Point(double x, double y) {
        this.x = x;
        this.y = y;
    }

    // المنشئ الافتراضي (يجب إضافته بعد إضافة المنشئ الصريح)
    public Point() {
        this.x = 0;
        this.y = 0;
    }
}

2 تمرين صنف Segment (قطعة مستقيمة)

الأسئلة:

  1. عرّف صنف Segment يمثل قطعة مستقيمة بنقطتين debut و fin
    • أضف منشئاً ذو معاملات
    • affiche(): عرض إحداثيات القطعة
    • longueur(): حساب طول القطعة
    • milieu(): إرجاع نقطة المنتصف
    • compareTo(Segment s): مقارنة قطعتين (0: متساويتان، 1: الأولى أكبر، -1: الثانية أكبر)
  2. أنشئ صنف ProgSegment ينشئ 100 قطعة، يرتبها تصاعدياً، ويعرضها
  3. عدّل الصنف لتكون كل القطعات بلون أحمر (استخدم static)

💻 حل التمرين 2 - الكود الكامل:

Segment.java 
public class Segment {
    // الخصائص
    private Point debut;
    private Point fin;
    private static String couleur = "rouge";  // مشترك بين كل الكائنات

    // المنشئ
    public Segment(Point d, Point f) {
        this.debut = d;
        this.fin = f;
    }

    // عرض القطعة
    public void affiche() {
        System.out.print("Segment [");
        debut.afficher();
        System.out.print(" -> ");
        fin.afficher();
        System.out.println(", Couleur: " + couleur + "]");
    }

    // حساب الطول
    public double longueur() {
        double dx = fin.getX() - debut.getX();
        double dy = fin.getY() - debut.getY();
        return Math.sqrt(dx*dx + dy*dy);
    }

    // نقطة المنتصف
    public Point milieu() {
        double mx = (debut.getX() + fin.getX()) / 2;
        double my = (debut.getY() + fin.getY()) / 2;
        return new Point(mx, my);
    }

    // مقارنة قطعتين
    public int compareTo(Segment s) {
        double l1 = this.longueur();
        double l2 = s.longueur();
        
        if (l1 == l2) return 0;
        else if (l1 > l2) return 1;
        else return -1;
    }

    // getter للون
    public static String getCouleur() {
        return couleur;
    }
}

1 صنف Vecteur (متجه)

المطلوب:

  1. عرّف صنف Vecteur للأعداد الصحيحة بخصائص: مرجع (référence) وحجم (taille)
  2. أضف: منشئ، طريقة remplir()، طريقة afficher()، وطريقة somme(Vecteur v)
  3. اكتب برنامجاً ينشئ متجهين ويحسب مجموعهما
Vecteur.java 
import java.util.Scanner;

public class Vecteur {
    private int[] reference;
    private int taille;

    // المنشئ
    public Vecteur(int t) {
        this.taille = t;
        this.reference = new int[t];
    }

    // ملء المتجه
    public void remplir(Scanner sc) {
        System.out.println("أدخل " + taille + " عنصر:");
        for (int i = 0; i < taille; i++) {
            reference[i] = sc.nextInt();
        }
    }

    // عرض المتجه
    public void afficher() {
        System.out.print("[ ");
        for (int i = 0; i < taille; i++) {
            System.out.print(reference[i]);
            if (i < taille - 1) System.out.print(", ");
        }
        System.out.println(" ]");
    }

    // حساب المجموع
    public Vecteur somme(Vecteur v) {
        if (this.taille != v.taille) {
            System.out.println("خطأ: المتجهان غير متساويين في الحجم!");
            return null;
        }
        
        Vecteur resultat = new Vecteur(taille);
        for (int i = 0; i < taille; i++) {
            resultat.reference[i] = this.reference[i] + v.reference[i];
        }
        return resultat;
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        System.out.print("أدخل حجم المتجهات: ");
        int n = sc.nextInt();
        
        Vecteur v1 = new Vecteur(n);
        Vecteur v2 = new Vecteur(n);
        
        v1.remplir(sc);
        v2.remplir(sc);
        
        System.out.print("\nالمتجه الأول: ");
        v1.afficher();
        
        System.out.print("المتجه الثاني: ");
        v2.afficher();
        
        Vecteur v3 = v1.somme(v2);
        System.out.print("المجموع:       ");
        v3.afficher();
    }
}

        return true;
    }

    // حذف وإرجاع العنصر العلوي
    public int depiler() {
        if (pileVide()) {
            System.out.println("الكومة فارغة!");
            return Integer.MIN_VALUE;
        }
        return elements[sommet--];
    }

    // إرجاع العنصر العلوي بدون حذف
    public int sommetPile() {
        if (pileVide()) {
            System.out.println("الكومة فارغة!");
            return Integer.MIN_VALUE;
        }
        return elements[sommet];
    }

    // اختبار الفراغ
    public boolean pileVide() {
        return sommet == -1;
    }

    // اختبار الامتلاء
    public boolean pilePleine() {
        return sommet == capacite - 1;
    }

    // عرض محتويات الكومة
    public void afficher() {
        System.out.print("[ ");
        for (int i = 0; i <= sommet; i++) {
            System.out.print(elements[i]);
            if (i < sommet) System.out.print(" | ");
        }
        System.out.println(" ] ← أعلى الكومة");
    }

    // حذف تكرارات قيمة معينة
    public void supprimerRepetitions(int val) {
        Pile temp = new Pile(capacite);
        boolean trouve = false;

        // نقل العناصر إلى كومة مؤقتة
        while (!pileVide()) {
            int elem = depiler();
            if (elem != val || !trouve) {
                temp.empiler(elem);
                if (elem == val) trouve = true;
            }
        }

        // إعادة العناصر إلى الكومة الأصلية
        while (!temp.pileVide()) {
            empiler(temp.depiler());
        }
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        System.out.print("أدخل عدد عناصر الكومة: ");
        int n = sc.nextInt();
        
        Pile pile = new Pile(n);
        
        System.out.println("أدخل العناصر:");
        for (int i = 0; i < n; i++) {
            pile.empiler(sc.nextInt());
        }
        
        System.out.print("\nالكومة الأصلية: ");
        pile.afficher();
        
        System.out.print("أدخل القيمة المراد حذف تكراراتها: ");
        int val = sc.nextInt();
        
        pile.supprimerRepetitions(val);
        
        System.out.print("بعد الحذف:      ");
        pile.afficher();
    }
}

4 صنف Intervalle (فترة)

المطلوب:

  1. نفذ صنف Intervalle بخصائص BI (البداية) و BS (النهاية)
  2. أضف الطرق:
    • afficher(): عرض [BI, BS]
    • milieu(): حساب نقطة المنتصف
    • avant(Intervalle i): هل الفترة قبل فترة أخرى؟
    • disjoint(Intervalle i): هل الفترتان منفصلتان؟
    • contenu(Intervalle i): هل الفترة داخل أخرى؟
    • intersection(Intervalle i): حساب التقاطع
    • fusion(Intervalle i): حساب الاتحاد
  3. أنشئ متجه من n فترة ورقمها تسلسلياً، ثم حدد تقاطع الكل
Intervalle.java 
public class Intervalle {
    private double BI;  // Borne Inférieure
    private double BS;  // Borne Supérieure
    private static int compteur = 0;
    private int numero;

    // المنشئ
    public Intervalle(double bi, double bs) {
        if (bi > bs) {
            double temp = bi;
            bi = bs;
            bs = temp;
        }
        this.BI = bi;
        this.BS = bs;
        this.numero = ++compteur;
    }

    // عرض الفترة
    public void afficher() {
        System.out.println("I" + numero + ": [" + BI + ", " + BS + "]");
    }

    // نقطة المنتصف
    public double milieu() {
        return (BI + BS) / 2;
    }

    // هل الفترة قبل فترة أخرى؟
    public boolean avant(Intervalle i) {
        return this.BS < i.BI;
    }

    // هل الفترتان منفصلتان؟
    public boolean disjoint(Intervalle i) {
        return (this.BS < i.BI || this.BI > i.BS);
    }

    // هل الفترة داخل أخرى؟
    public boolean contenu(Intervalle i) {
        return (this.BI >= i.BI && this.BS <= i.BS);
    }

    // حساب التقاطع
    public Intervalle intersection(Intervalle i) {
        if (this.disjoint(i)) {
            System.out.println("لا يوجد تقاطع!");
            return null;
        }
        double newBI = Math.max(this.BI, i.BI);
        double newBS = Math.min(this.BS, i.BS);
        return new Intervalle(newBI, newBS);
    }

    // حساب الاتحاد
    public Intervalle fusion(Intervalle i) {
        if (this.disjoint(i)) {
            System.out.println("لا يمكن الاتحاد: الفترتان منفصلتان!");
            return null;
        }
        double newBI = Math.min(this.BI, i.BI);
        double newBS = Math.max(this.BS, i.BS);
        return new Intervalle(newBI, newBS);
    }

    // getters
    public double getBI() { return BI; }
    public double getBS() { return BS; }

    // برنامج الاختبار
    public static void main(String[] args) {
        // إنشاء 5 فترات
        Intervalle[] intervalles = {
            new Intervalle(1, 5),
            new Intervalle(3, 8),
            new Intervalle(4, 10),
            new Intervalle(2, 6),
            new Intervalle(5, 9)
        };

        System.out.println("=== الفترات المُنشأة ===");
        for (Intervalle iv : intervalles) {
            iv.afficher();
        }

        // حساب تقاطع كل الفترات
        Intervalle result = intervalles[0];
        for (int i = 1; i < intervalles.length; i++) {
            result = result.intersection(intervalles[i]);
            if (result == null) {
                System.out.println("لا يوجد تقاطع عام!");
                return;
            }
        }

        System.out.println("\n=== تقاطع كل الفترات ===");
        result.afficher();
    }
}

1 صنف Calcul (حسابات ثابتة)

المطلوب:

  1. أنشئ صنف Calcul يحتوي على طرق ثابتة (static):
    • factoriel(n): حساب $n!$
    • puissance(x, y): حساب $x^y$
    • sommeN2(n): مجموع أول $n$ أعداد زوجية
  2. في صنف ProgCalcul، أنشئ متجه حجمه n يحتوي في كل عنصر i على: factoriel(i), puissance(i,2), sommeN2(i)
Calcul.java 
public class Calcul {

    // حساب المضروب n!
    public static long factoriel(int n) {
        if (n < 0) return -1;
        long result = 1;
        for (int i = 2; i <= n; i++) {
            result *= i;
        }
        return result;
    }

    // حساب الأس x^y
    public static long puissance(int x, int y) {
        long result = 1;
        for (int i = 0; i < y; i++) {
            result *= x;
        }
        return result;
    }

    // مجموع أول n أعداد زوجية
    public static int sommeN2(int n) {
        int sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += (2 * i);
        }
        return sum;
    }
}

3 صنف Monome و Polynome (متعدد الحدود)

المطلوب:

  1. صنف Monome: coefficient (معامل), exposant (أس)
    • منشئ، lire()، afficher() (مثال: $-4x^3$)
    • somme(Monome m): جمع حدود same degree
    • valeur(double x): حساب القيمة عند x
  2. صنف Polynome: degré (درجة), tableau de Monômes
    • منشئ، lire()، afficher()
    • somme(Poly p): جمع متعددي حدود
    • valeur(double x): حساب القيمة
  3. برنامج ينشئ P1 و P2 ويحسب المجموع والقيمة
Monome.java 
import java.util.Scanner;

public class Monome {
    private double coefficient;
    private int exposant;

    // المنشئ
    public Monome(double coef, int exp) {
        this.coefficient = coef;
        this.exposant = exp;
    }

    // المنشئ الافتراضي
    public Monome() {}

    // قراءة الحد
    public void lire(Scanner sc) {
        System.out.print("المعامل: ");
        coefficient = sc.nextDouble();
        System.out.print("الأس: ");
        exposant = sc.nextInt();
    }

    // عرض الحد (مثال: -4x^3 أو 5x^2 أو 3)
    public void afficher() {
        if (exposant == 0) {
            System.out.print(coefficient);
        } } else if (exposant == 1) {
            System.out.print(coefficient + "x");
        } else {
            System.out.print(coefficient + "x^" + exposant);
        }
    }

    // جمع حدان من نفس الدرجة
    public Monome somme(Monome m) {
        if (this.exposant != m.exposant) {
            System.out.println("خطأ: لا يمكن جمع حدود من درجات مختلفة!");
            return null;
        }
        return new Monome(this.coefficient + m.coefficient, exposant);
    }

    // حساب القيمة عند x
    public double valeur(double x) {
        return coefficient * Math.pow(x, exposant);
    }

    // getters
    public double getCoefficient() { return coefficient; }
    public int getExposant() { return exposant; }
}
Polynome.java 
public class Polynome {
    private int degre;
    private Monome[] monomes;

    // المنشئ
    public Polynome(int deg) {
        this.degre = deg;
        this.monomes = new Monome[deg + 1];
    }

    // قراءة متعدد الحدود
    public void lire(java.util.Scanner sc) {
        System.out.println("أدخل حدود متعدد الحدود (من الأعلى إلى الأدنى):");
        for (int i = degre; i >= 0; i--) {
            System.out.println("الحد من الدرجة " + i + ":");
            monomes[i] = new Monome();
            monomes[i].lire(sc);
        }
    }

    // عرض متعدد الحدود
    public void afficher() {
        System.out.print("P(x) = ");
        boolean premier = true;
        
        for (int i = degre; i >= 0; i--) {
            if (monomes[i].getCoefficient() != 0) {
                if (!premier && monomes[i].getCoefficient() > 0) {
                    System.out.print(" + ");
                } else if (!premier) {
                    System.out.print(" ");
                }
                monomes[i].afficher();
                premier = false;
            }
        }
        System.out.println();
    }

    // حساب القيمة عند x
    public double valeur(double x) {
        double result = 0;
        for (int i = 0; i <= degre; i++) {
            result += monomes[i].valeur(x);
        }
        return result;
    }

    // جمع متعددي حدود
    public Polynome somme(Polynome p) {
        int newDeg = Math.max(this.degre, p.degre);
        Polynome result = new Polynome(newDeg);
        
        for (int i = 0; i <= newDeg; i++) {
            double coef1 = 0, coef2 = 0;
            
            if (i <= this.degre) coef1 = this.monomes[i].getCoefficient();
            if (i <= p.degre) coef2 = p.monomes[i].getCoefficient();
            
            result.monomes[i] = new Monome(coef1 + coef2, i);
        }
        
        return result;
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        
        // إنشاء P1
        System.out.print("درجة P1: ");
        int n1 = sc.nextInt();
        Polynome P1 = new Polynome(n1);
        P1.lire(sc);
        
        // إنشاء P2
        System.out.print("درجة P2: ");
        int n2 = sc.nextInt();
        Polynome P2 = new Polynome(n2);
        P2.lire(sc);

        System.out.println("\nP1 = ");
        P1.afficher();
        
        System.out.println("P2 = ");
        P2.afficher();

        // حساب المجموع
        Polynome PSomme = P1.somme(P2);
        System.out.println("\nP1 + P2 = ");
        PSomme.afficher();

        // حساب القيمة
        System.out.print("\nأدخل قيمة x: ");
        double x = sc.nextDouble();
        System.out.printf("P(%s) = %.2f\n", x, PSomme.valeur(x));
    }
}

4 صنف Etudiant (طالب) مع Mention

المطلوب:

  1. صنف Etudiant: matricule, nom, prénom, 3 modules (C1,C2,C3), 3 notes (N1,N2,N3)
    • Lire(): قراءة البيانات
    • AttribuerNotes(): إدخال الملاحظات
    • CalculerMoy(): حساب المعدل المرجح
    • Afficher(): عرض مع Mention
  2. Mention: Excellent(≥18), Très bien(≥16), Bien(≥14), Assez bien(≥12), Passable(≥11)
  3. برنامج يخزن n طلاب ويعرضهم مع المعدلات والمذكرات
Etudiant.java 
import java.util.Scanner;

// تعداد المذكرات
enum Mention {
    EXCELLENT("Excellent", 18),
    TRES_BIEN("Très Bien", 16),
    BIEN("Bien", 14),
    ASSEZ_BIEN("Assez Bien", 12),
    PASSABLE("Passable", 11),
    AJOURNE("Ajourné", 0);

    private String label;
    private int seuil;

    private Mention(String label, int seuil) {
        this.label = label;
        this.seuil = seuil;
    }

    public String getLabel() { return label; }
    public int getSeuil() { return seuil; }

    public static Mention getMention(double moyenne) {
        if (moyenne >= EXCELLENT.seuil) return EXCELLENT;
        if (moyenne >= TRES_BIEN.seuil) return TRES_BIEN;
        if (moyenne >= BIEN.seuil) return BIEN;
        if (moyenne >= ASSEZ_BIEN.seuil) return ASSEZ_BIEN;
        if (moyenne >= PASSABLE.seuil) return PASSABLE;
        return AJOURNE;
    }
}

public class Etudiant {
    private long matricule;
    private String nom;
    private String prenom;
    private double C1, C2, C3;  // المعاملات
    private double N1, N2, N3;  // الملاحظات
    private double moyenne;
    private Mention mention;

    // قراءة البيانات الشخصية
    public void Lire(Scanner sc) {
        System.out.print("الماتريكول: ");
        matricule = sc.nextLong();
        sc.nextLine();
        
        System.out.print("الاسم: ");
        nom = sc.nextLine();
        
        System.out.print("الاسم الأول: ");
        prenom = sc.nextLine();

        // المعاملات الافتراضية
        C1 = 3; C2 = 2; C3 = 1;
    }

    // إدخال الملاحظات
    public void AttribuerNotes(Scanner sc) {
        System.out.println("\n--- ملاحظات الطالب " + nom + " ---");
        System.out.print("ملاحظة المادة 1 (معامل " + C1 + "): ");
        N1 = sc.nextDouble();
        System.out.print("ملاحظة المادة 2 (معامل " + C2 + "): ");
        N2 = sc.nextDouble();
        System.out.print("ملاحظة المادة 3 (معامل " + C3 + "): ");
        N3 = sc.nextDouble();
    }

    // حساب المعدل المرجح
    public void CalculerMoy() {
        double totalCoeff = C1 + C2 + C3;
        moyenne = (N1*C1 + N2*C2 + N3*C3) / totalCoeff;
        mention = Mention.getMention(moyenne);
    }

    // عرض البيانات
    public void Afficher() {
        System.out.println("╔══════════════════════════════════════╗");
        System.out.printf("║ الماتريكول: %-24d ║\n", matricule);
        System.out.printf("║ الاسم: %-29s ║\n", nom + " " + prenom);
        System.out.printf("║ الملاحظات: %.2f | %.2f | %.2f       ║\n", N1, N2, N3);
        System.out.printf("║ المعدل: %-29.2f ║\n", moyenne);
        System.out.printf("║ المذكرة: %-28s ║\n", mention.getLabel());
        System.out.println("╚══════════════════════════════════════╝");
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        System.out.print("عدد الطلاب: ");
        int n = sc.nextInt();
        
        Etudiant[] etudiants = new Etudiant[n];
        
        // إدخال البيانات
        for (int i = 0; i < n; i++) {
            System.out.println("\n=== الطالب رقم " + (i+1) + " ===");
            etudiants[i] = new Etudiant();
            etudiants[i].Lire(sc);
            etudiants[i].AttribuerNotes(sc);
            etudiants[i].CalculerMoy();
        }

        // عرض النتائج
        System.out.println("\n\n========================================");
        System.out.println("       قائمة الطلاب والمعدلات");
        System.out.println("========================================\n");
        
        for (int i = 0; i < n; i++) {
            etudiants[i].Afficher();
            System.out.println();
        }
    }
}
