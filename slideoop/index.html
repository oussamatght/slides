Point.java 
public class Point {
    // الخصائص (Attributs)
    private double x;
    private double y;
    
    // المنشئ الافتراضي (Constructeur par défaut)
    public Point() {
        this.x = 0;
        this.y = 0;
    }
    
    // المنشئ ذو المعاملات (Constructeur paramétré)
    public Point(double x, double y) {
        this.x = x;
        this.y = y;
    }
    
    // طرق الوصول (Accesseurs - Getters)
    public double getX() { return this.x; }
    public double getY() { return this.y; }
    
    // طرق التعديل (Mutateurs - Setters)
    public void setX(double x) { this.x = x; }
    public void setY(double y) { this.y = y; }
    
    // طريقة النقل
    public void deplacer(double a, double b) {
        this.x += a;
        this.y += b;
    }
    
    // طريقة العرض
    public void afficher() {
        System.out.println("(" + x + ", " + y + ")");
    }
}

    private Point centre;
    private double rayon;

    public Cercle() {

    // المنشئ الثاني (التزيين - Surcharge)

        this.rayon = r;
        this.couleur = c;
    }
    
    // طريقة النقل
    public void deplacer(double a, double b) {
        this.centre.deplacer(a, b);
    }
    
    // حساب المساحة
    public double surface() {
        return Math.PI * rayon * rayon;
    }
}

Point.java 
public class Point {
    // الخصائص
    private double x;
    private double y;

    // Getters
    public double getX() { return x; }
    public double getY() { return y; }

    // Setters
    public void setX(double x) { this.x = x; }
    public void setY(double y) { this.y = y; }

    // عرض الإحداثيات
    public void afficher() {
        System.out.println("(" + x + ", " + y + ")");
    }

    // حساب النقطة الوسطى
    public Point milieu(Point p) {
        Point m = new Point();
        m.setX((x + p.x) / 2);
        m.setY((y + p.y) / 2);
        return m;
    }

    // نظير النقطة بالنسبة للأصل
    public Point symetrie() {
        return new Point(-x, -y);
    }

    // اختبار التساوي
    public boolean egal(Point p) {
        return (x == p.x && y == p.y);
    }

    // المقارنة بالقرب من المركز
    public Point procheDuCentre(Point p) {
        double dist1 = Math.sqrt(x*x + y*y);
        double dist2 = Math.sqrt(p.x*p.x + p.y*p.y);
        if (dist1 <= dist2)
            return this;
        else
            return p;
    }

    // المنشئ الصريح (يُضاف لاحقاً)
    public Point(double x, double y) {
        this.x = x;
        this.y = y;
    }

    // المنشئ الافتراضي (يجب إضافته بعد إضافة المنشئ الصريح)
    public Point() {
        this.x = 0;
        this.y = 0;
    }
}

الأسئلة:

  1. عرّف صنف Segment يمثل قطعة مستقيمة بنقطتين debut و fin

Segment.java 
public class Segment {
    // الخصائص
    private Point debut;
    private Point fin;
    private static String couleur = "rouge";  // مشترك بين كل الكائنات

    // المنشئ
    public Segment(Point d, Point f) {
        this.debut = d;
        this.fin = f;
    }

    // عرض القطعة
    public void affiche() {
        System.out.print("Segment [");
        debut.afficher();
        System.out.print(" -> ");
        fin.afficher();
        System.out.println(", Couleur: " + couleur + "]");
    }

    // حساب الطول
    public double longueur() {
        double dx = fin.getX() - debut.getX();
        double dy = fin.getY() - debut.getY();
        return Math.sqrt(dx*dx + dy*dy);
    }

    // نقطة المنتصف
    public Point milieu() {
        double mx = (debut.getX() + fin.getX()) / 2;
        double my = (debut.getY() + fin.getY()) / 2;
        return new Point(mx, my);
    }

    // مقارنة قطعتين
    public int compareTo(Segment s) {
        double l1 = this.longueur();
        double l2 = s.longueur();
        
        if (l1 == l2) return 0;
        else if (l1 > l2) return 1;
        else return -1;
    }

    // getter للون
    public static String getCouleur() {
        return couleur;
    }
}

1 صنف Vecteur (متجه)

المطلوب:

  1. عرّف صنف Vecteur للأعداد الصحيحة بخصائص: مرجع (référence) وحجم (taille)
  2. أضف: منشئ، طريقة remplir()، طريقة afficher()، وطريقة somme(Vecteur v)
  3. اكتب برنامجاً ينشئ متجهين ويحسب مجموعهما
Vecteur.java 
import java.util.Scanner;

public class Vecteur {
    private int[] reference;
    private int taille;

    // المنشئ
    public Vecteur(int t) {
        this.taille = t;
        this.reference = new int[t];
    }

    // ملء المتجه
    public void remplir(Scanner sc) {
        System.out.println("أدخل " + taille + " عنصر:");
        for (int i = 0; i < taille; i++) {
            reference[i] = sc.nextInt();
        }
    }

    // عرض المتجه
    public void afficher() {
        System.out.print("[ ");
        for (int i = 0; i < taille; i++) {
            System.out.print(reference[i]);
            if (i < taille - 1) System.out.print(", ");
        }
        System.out.println(" ]");
    }

    // حساب المجموع
    public Vecteur somme(Vecteur v) {
        if (this.taille != v.taille) {
            System.out.println("خطأ: المتجهان غير متساويين في الحجم!");
            return null;
        }
        
        Vecteur resultat = new Vecteur(taille);
        for (int i = 0; i < taille; i++) {
            resultat.reference[i] = this.reference[i] + v.reference[i];
        }
        return resultat;
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        System.out.print("أدخل حجم المتجهات: ");
        int n = sc.nextInt();
        
        Vecteur v1 = new Vecteur(n);
        Vecteur v2 = new Vecteur(n);
        
        v1.remplir(sc);
        v2.remplir(sc);
        
        System.out.print("\nالمتجه الأول: ");
        v1.afficher();
        
        System.out.print("المتجه الثاني: ");
        v2.afficher();
        
        Vecteur v3 = v1.somme(v2);
        System.out.print("المجموع:       ");
        v3.afficher();
    }
}

    // حذف وإرجاع العنصر العلوي
    public int depiler() {
        if (pileVide()) {
            System.out.println("الكومة فارغة!");
            return Integer.MIN_VALUE;
        }
        return elements[sommet--];
    }

    // إرجاع العنصر العلوي بدون حذف
    public int sommetPile() {
        if (pileVide()) {
            System.out.println("الكومة فارغة!");
            return Integer.MIN_VALUE;
        }
        return elements[sommet];
    }

    // اختبار الفراغ
    public boolean pileVide() {
        return sommet == -1;
    }

    // اختبار الامتلاء
    public boolean pilePleine() {
        return sommet == capacite - 1;
    }

    // عرض محتويات الكومة
    public void afficher() {
        System.out.print("[ ");
        for (int i = 0; i <= sommet; i++) {
            System.out.print(elements[i]);
            if (i < sommet) System.out.print(" | ");
        }
        System.out.println(" ] ← أعلى الكومة");
    }

    // حذف تكرارات قيمة معينة
    public void supprimerRepetitions(int val) {
        Pile temp = new Pile(capacite);
        boolean trouve = false;

        // نقل العناصر إلى كومة مؤقتة
        while (!pileVide()) {
            int elem = depiler();
            if (elem != val || !trouve) {
                temp.empiler(elem);
                if (elem == val) trouve = true;
            }
        }

        // إعادة العناصر إلى الكومة الأصلية
        while (!temp.pileVide()) {
            empiler(temp.depiler());
        }
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        System.out.print("أدخل عدد عناصر الكومة: ");
        int n = sc.nextInt();
        
        Pile pile = new Pile(n);
        
        System.out.println("أدخل العناصر:");
        for (int i = 0; i < n; i++) {
            pile.empiler(sc.nextInt());
        }
        
        System.out.print("\nالكومة الأصلية: ");
        pile.afficher();
        
        System.out.print("أدخل القيمة المراد حذف تكراراتها: ");
        int val = sc.nextInt();
        
        pile.supprimerRepetitions(val);
        
        System.out.print("بعد الحذف:      ");
        pile.afficher();
    }
}

4 صنف Intervalle (فترة)

المطلوب:

  1. نفذ صنف Intervalle بخصائص BI (البداية) و BS (النهاية)
  2. أضف الطرق:
    • afficher(): عرض [BI, BS]
    • milieu(): حساب نقطة المنتصف
    • avant(Intervalle i): هل الفترة قبل فترة أخرى؟
    • disjoint(Intervalle i): هل الفترتان منفصلتان؟
    • contenu(Intervalle i): هل الفترة داخل أخرى؟
    • intersection(Intervalle i): حساب التقاطع
    • fusion(Intervalle i): حساب الاتحاد
  3. أنشئ متجه من n فترة ورقمها تسلسلياً، ثم حدد تقاطع الكل
Intervalle.java 
public class Intervalle {
    private double BI;  // Borne Inférieure
    private double BS;  // Borne Supérieure
    private static int compteur = 0;
    private int numero;

    // المنشئ
    public Intervalle(double bi, double bs) {
        if (bi > bs) {
            double temp = bi;
            bi = bs;
            bs = temp;
        }
        this.BI = bi;
        this.BS = bs;
        this.numero = ++compteur;
    }

    // عرض الفترة
    public void afficher() {
        System.out.println("I" + numero + ": [" + BI + ", " + BS + "]");
    }

    // نقطة المنتصف
    public double milieu() {
        return (BI + BS) / 2;
    }

    // هل الفترة قبل فترة أخرى؟
    public boolean avant(Intervalle i) {
        return this.BS < i.BI;
    }

    // هل الفترتان منفصلتان؟
    public boolean disjoint(Intervalle i) {
        return (this.BS < i.BI || this.BI > i.BS);
    }

    // هل الفترة داخل أخرى؟
    public boolean contenu(Intervalle i) {
        return (this.BI >= i.BI && this.BS <= i.BS);
    }

    // حساب التقاطع
    public Intervalle intersection(Intervalle i) {
        if (this.disjoint(i)) {
            System.out.println("لا يوجد تقاطع!");
            return null;
        }
        double newBI = Math.max(this.BI, i.BI);
        double newBS = Math.min(this.BS, i.BS);
        return new Intervalle(newBI, newBS);
    }

    // حساب الاتحاد
    public Intervalle fusion(Intervalle i) {
        if (this.disjoint(i)) {
            System.out.println("لا يمكن الاتحاد: الفترتان منفصلتان!");
            return null;
        }
        double newBI = Math.min(this.BI, i.BI);
        double newBS = Math.max(this.BS, i.BS);
        return new Intervalle(newBI, newBS);
    }

    // getters
    public double getBI() { return BI; }
    public double getBS() { return BS; }

    // برنامج الاختبار
    public static void main(String[] args) {
        // إنشاء 5 فترات
        Intervalle[] intervalles = {
            new Intervalle(1, 5),
            new Intervalle(3, 8),
            new Intervalle(4, 10),
            new Intervalle(2, 6),
            new Intervalle(5, 9)
        };

        System.out.println("=== الفترات المُنشأة ===");
        for (Intervalle iv : intervalles) {
            iv.afficher();
        }

        // حساب تقاطع كل الفترات
        Intervalle result = intervalles[0];
        for (int i = 1; i < intervalles.length; i++) {
            result = result.intersection(intervalles[i]);
            if (result == null) {
                System.out.println("لا يوجد تقاطع عام!");
                return;
            }
        }

        System.out.println("\n=== تقاطع كل الفترات ===");
        result.afficher();
    }
}

public class Calcul {

    // حساب المضروب n!
    public static long factoriel(int n) {
        if (n < 0) return -1;
        long result = 1;
        for (int i = 2; i <= n; i++) {
            result *= i;
        }
        return result;
    }

    // حساب الأس x^y
    public static long puissance(int x, int y) {
        long result = 1;
        for (int i = 0; i < y; i++) {
            result *= x;
        }
        return result;
    }

    // مجموع أول n أعداد زوجية
    public static int sommeN2(int n) {
        int sum = 0;
        for (int i = 1; i <= n; i++) {
            sum += (2 * i);
        }
        return sum;
    }
}

import java.util.Scanner;

public class Monome {
    private double coefficient;

        return coefficient * Math.pow(x, exposant);

public class Polynome {
    private int degre;
    private Monome[] monomes;

    // المنشئ
    public Polynome(int deg) {
        this.degre = deg;
        this.monomes = new Monome[deg + 1];
    }

    // قراءة متعدد الحدود
    public void lire(java.util.Scanner sc) {
        System.out.println("أدخل حدود متعدد الحدود (من الأعلى إلى الأدنى):");
        for (int i = degre; i >= 0; i--) {
            System.out.println("الحد من الدرجة " + i + ":");
            monomes[i] = new Monome();
            monomes[i].lire(sc);
        }
    }

    // عرض متعدد الحدود
    public void afficher() {
        System.out.print("P(x) = ");
        boolean premier = true;
        
        for (int i = degre; i >= 0; i--) {
            if (monomes[i].getCoefficient() != 0) {
                if (!premier && monomes[i].getCoefficient() > 0) {
                    System.out.print(" + ");
                } else if (!premier) {
                    System.out.print(" ");
                }
                monomes[i].afficher();
                premier = false;
            }
        }
        System.out.println();
    }

    // حساب القيمة عند x
    public double valeur(double x) {
        double result = 0;
        for (int i = 0; i <= degre; i++) {
            result += monomes[i].valeur(x);
        }
        return result;
    }

    // جمع متعددي حدود
    public Polynome somme(Polynome p) {
        int newDeg = Math.max(this.degre, p.degre);
        Polynome result = new Polynome(newDeg);
        
        for (int i = 0; i <= newDeg; i++) {
            double coef1 = 0, coef2 = 0;
            
            if (i <= this.degre) coef1 = this.monomes[i].getCoefficient();
            if (i <= p.degre) coef2 = p.monomes[i].getCoefficient();
            
            result.monomes[i] = new Monome(coef1 + coef2, i);
        }
        
        return result;
    }

    // برنامج الاختبار
    public static void main(String[] args) {
        java.util.Scanner sc = new java.util.Scanner(System.in);
        
        // إنشاء P1
        System.out.print("درجة P1: ");
        int n1 = sc.nextInt();
        Polynome P1 = new Polynome(n1);
        P1.lire(sc);
        
        // إنشاء P2
        System.out.print("درجة P2: ");
        int n2 = sc.nextInt();
        Polynome P2 = new Polynome(n2);
        P2.lire(sc);

        System.out.println("\nP1 = ");
        P1.afficher();
        
        System.out.println("P2 = ");
        P2.afficher();

        // حساب المجموع
        Polynome PSomme = P1.somme(P2);
        System.out.println("\nP1 + P2 = ");
        PSomme.afficher();

        // حساب القيمة
        System.out.print("\nأدخل قيمة x: ");
        double x = sc.nextDouble();
        System.out.printf("P(%s) = %.2f\n", x, PSomme.valeur(x));
    }
}

import java.util.Scanner;

// تعداد المذكرات
enum Mention {
    EXCELLENT("Excellent", 18),
    TRES_BIEN("Très Bien", 16),
    BIEN("Bien", 14),
    ASSEZ_BIEN("Assez Bien", 12),
    PASSABLE("Passable", 11),
    AJOURNE("Ajourné", 0);

    private String label;
    private int seuil;

    private Mention(String label, int seuil) {
        this.label = label;
        this.seuil = seuil;
    }

    public String getLabel() { return label; }
    public int getSeuil() { return seuil; }

    public static Mention getMention(double moyenne) {
        if (moyenne >= EXCELLENT.seuil) return EXCELLENT;
        if (moyenne >= TRES_BIEN.seuil) return TRES_BIEN;
        if (moyenne >= BIEN.seuil) return BIEN;
        if (moyenne >= ASSEZ_BIEN.seuil) return ASSEZ_BIEN;
        if (moyenne >= PASSABLE.seuil) return PASSABLE;
        return AJOURNE;
    }
}

🎯 الفرق: Abstract Class vs Interface
  • Abstract Class: قد تحتوي كود حقيقي (مثل Afficher() في Forme)، وراثة أب واحد فقط، متغيرات protected/private

  • متى تستخدم؟ فيه كود مشترك حقيقي؟ → abstract class. فقط السلوك المتوقع؟ → interface

🏛️ المفاهيم الأساسية في البرمجة كائنية التوجه

هذا الجزء يغطي جميع المفاهيم الأساسية التي تحتاج إلى فهمها بعمق لإتقان البرمجة كائنية التوجه في Java.

1️⃣ المفاهيم الأساسية

الصنف (Class)

تعريف: مجموعة من البيانات (متغيرات) + الطرق (functions) التي تعمل عليها

المكونات:

  • الخصائص (Attributes)
  • الطرق (Methods)
  • المنشئات (Constructors)

الكائن (Object)

تعريف: نسخة ملموسة من صنف ما، يحتوي على قيم حقيقية للخصائص

الفرق:

  • الصنف = قالب عام
  • الكائن = حالة محددة من الصنف

المنشئ (Constructor)

طريقة خاصة بنفس اسم الصنف، بدون نوع إرجاع، تستخدم لتهيئة قيم الكائن

خصائص المنشئ:

  • نفس الاسم تماماً كالصنف
  • بدون نوع إرجاع (ليس حتى void)
  • يمكن تعريف عدة منشئات (Overloading)
  • يُستدعى تلقائياً عند إنشاء كائن جديد

الكلمة المحفوظة this

تُستخدم للإشارة إلى الكائن الحالي داخل الطريقة. تساعد في التمييز بين:

  • this.nom = الخاصية (متغير الصنف)
  • nom = المعامل (parameter)

ملاحظة: لا يمكن استخدام this في main

2️⃣ مجالات الرؤية والوصول

مستويات الرؤية للطرق والخصائص (4 مستويات):

المستوى الكلمة المحفوظة الوصول
🔓 الأعلى public متاح من جميع الأصناف والحزم
🔓 متوسط protected الحزمة + الأبناء (الأصناف الموروثة)
🔒 منخفض بدون كلمة الحزمة فقط (package)
🔐 الأقل private الصنف نفسه فقط

Getters و Setters

Getter: طريقة للوصول إلى قيمة خاصية private

public double getSolde() { return solde; }

Setter: طريقة لتعديل قيمة خاصية private

public void setSolde(double s) { solde = s; }

3️⃣ Static و Final

المتغير الثابت (Static Variable)

متغير مشترك بين جميع الكائنات من نفس الصنف

مثال: عداد يحسب عدد الكائنات المُنشأة

الوصول: ClassName.variableName

الملاحظة: يمكن الوصول إليها بدون إنشاء كائن

الطريقة الثابتة (Static Method)

طريقة مشتركة بين جميع الكائنات، مستقلة عن الكائن

مثال: Math.sqrt()

الوصول: ClassName.methodName()

القيد: لا تستطيع استخدام متغيرات instance

Final - الثابتة

ثابتة صنف: قيمة ثابتة لا تتغير للصنف كله

public static final double PI = 3.14;

ثابتة instance: قيمة ثابتة لا تتغير لكل كائن

طريقة final: لا يمكن إعادة تعريفها في الأبناء

4️⃣ الوراثة (Inheritance)

التعريف

آلية تسمح لصنف (الابن) بوراثة خصائص وطرق من صنف آخر (الأب)

الكلمة المحفوظة: extends

مثال: class Dog extends Animal

إعادة تعريف الطريقة (Method Overriding)

عندما يعيّن الابن تعريف طريقة من الأب بطريقة مختلفة

مثال:

class Animal { public void sound() { System.out.println("..."); } }
class Dog extends Animal {
  public void sound() { System.out.println("Woof!"); } // Override
}

الكلمة super

تُستخدم للإشارة إلى الصنف الأب

  • super.methodName() = استدعاء طريقة من الأب
  • super() = استدعاء منشئ الأب

5️⃣ Casting (التحويل)

Upcasting

تحويل من الابن إلى الأب (تحويل آمن وتلقائي)

Animal a = new Dog(); ✓ صحيح

الفكرة: الابن يدخل في الأب دائماً

Downcasting

تحويل من الأب إلى الابن (يحتاج casting واضح)

Dog d = (Dog) a; يحتاج cast

التحذير: قد تحدث أخطاء إذا لم يكن a فعلاً Dog

instanceof - التحقق من النوع

عامل للتحقق من كون الكائن نسخة من صنف معين

if (a instanceof Dog) { ... }

الاستخدام: تجنب الأخطاء قبل downcasting

6️⃣ التعددية (Polymorphism)

المفهوم

القدرة على استدعاء نفس الطريقة على أنواع مختلفة من الكائنات والحصول على تصرفات مختلفة

مثال عملي

Animal[] animals = {new Dog(), new Cat(), new Bird()};
for (Animal a : animals) {
  a.sound(); // كل واحد يُصدر صوتاً مختلفاً
}
الإخراج:
Woof! (Dog)
Meow! (Cat)
Tweet! (Bird)

7️⃣ الأصناف المجردة والواجهات

الصنف المجرد (Abstract Class)

صنف لا يمكن إنشاء كائنات منه مباشرة، يحتوي على طرق مجردة (بدون تنفيذ)

  • تُستخدم عندما يكون هناك كود مشترك حقيقي
  • الصنف يمكن أن يكون أب واحد فقط
  • الطرق المجردة يجب أن يُعاد تعريفها في الأبناء
  • مثال: abstract class Shape { abstract void draw(); }

الواجهة (Interface)

عقد يحدد السلوك المتوقع من الأصناف التي تطبقه، بدون تنفيذ فعلي

  • تحتوي على توقيعات الطرق فقط
  • فئة واحدة قد تطبق عدة واجهات
  • implements = تطبيق الواجهة
  • مثال: interface Comparable { int compareTo(Object o); }

الفرق بين Abstract و Interface

الميزة Abstract Class Interface
الوراثة أب واحد فقط عدة واجهات
الكود يمكن أن يحتوي كود توقيعات فقط
المتغيرات أي مستوى رؤية public final فقط
متى تُستخدم كود مشترك حقيقي سلوك متوقع فقط

📋 القوانين العامة للتصميم الجيد

1. الخصائص (Attributes)

✓ يجب أن تكون private أو protected

السبب: احترام مبدأ التغليف (Encapsulation)

2. الطرق (Methods)

✓ يجب أن تكون public أو protected

السبب: تشجيع إعادة الاستخدام (Reusability)

3. الأصناف (Classes)

✓ يجب أن تكون public

السبب: السماح باستخدامها من حزم أخرى

⚠️ ملاحظة هامة

ملف Java واحد يمكنه أن يحتوي على صنف public واحد فقط، وهذا الصنف يجب أن يحمل نفس اسم الملف

مثال: public class Dog { ... } → اسم الملف: Dog.java

✍️ اتفاقيات التسمية (Naming Conventions)

  • الأصناف: تبدأ بحرف كبير (PascalCase) مثل: Dog, Animal
  • الخصائص والطرق: تبدأ بحرف صغير (camelCase) مثل: getName(), age
  • الثوابت: كل الأحرف كبيرة مثل: PI, MAX_SIZE
  • الحزم: كل الأحرف صغيرة مثل: com.example.utilities
